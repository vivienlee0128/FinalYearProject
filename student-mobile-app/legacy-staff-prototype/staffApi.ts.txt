/**
 * Staff-facing frontend contract for n8n. Qwickly credentials never belong in Expo.
 * All live routes MUST validate Entra access tokens, staff role, and course access in n8n.
 * Until university approval and n8n endpoints exist, this module returns *clearly marked*
 * fictional demo fixtures. Failed live calls never silently fall back to synthetic data.
 */
export type Source = 'demo' | 'qwickly';
export type CaseStatus = 'open' | 'in_review' | 'resolved';
export type Priority = 'high' | 'medium' | 'low';
export type StaffCase = {
  id: string; sisId: string; studentName: string; reason: string;
  priority: Priority; status: CaseStatus; owner: string;
  createdAt: string; note: string;
};
export type StudentSummary = {
  sisId: string; name: string; programme: string; attendance: number | null;
  attended: number; scheduled: number; risk: 'review' | 'monitor' | 'none';
};
export type AuditIssue = {
  id: string; type: 'Missing session' | 'Identifier mismatch' | 'Sync failure' | 'Incomplete data';
  course: string; summary: string; severity: Priority; raisedAt: string; status: 'open' | 'resolved';
};
export type DashboardData = {
  source: Source;
  updatedAt: string;
  studentCount: number;
  meanAttendance: number | null;
  reportCount: number;
  trend: { label: string; percentage: number }[];
  courses: { code: string; title: string; percentage: number; students: number }[];
};
export type StaffApiConfig = { enabled: boolean; reason: string };

const STAFF_BASE = (process.env.EXPO_PUBLIC_N8N_STAFF_API_BASE_URL ?? '').trim().replace(/\/+$/, '');
const API_SCOPE = (process.env.EXPO_PUBLIC_MS_API_SCOPE ?? '').trim();
const STAFF_SCOPE_ENABLED = process.env.EXPO_PUBLIC_ENABLE_STAFF_API_SCOPE === 'true';

export function staffApiConfig(): StaffApiConfig {
  if (!STAFF_BASE) return { enabled: false, reason: 'Staff API not configured — showing fictional demonstration data.' };
  if (!/^https:\/\/[^\s/]+\/webhook\//i.test(STAFF_BASE))
    return { enabled: false, reason: 'Staff API must use a HTTPS n8n webhook URL.' };
  if (!STAFF_SCOPE_ENABLED) return { enabled: false, reason: 'Live staff API is disabled until the delegated scope is approved; showing demonstration data.' };
  if (!API_SCOPE || API_SCOPE.includes('YOUR_'))
    return { enabled: false, reason: 'A delegated Microsoft API scope must be approved before sending tokens to n8n.' };
  return { enabled: true, reason: 'Connected mode requested; n8n must enforce staff and course authorization.' };
}

async function request<T>(
  path: string,
  getAccessToken: () => Promise<string>,
  options: {method?: 'GET' | 'POST'; body?: unknown; signal?: AbortSignal} = {},
): Promise<T> {
  const configuration = staffApiConfig();
  if (!configuration.enabled) throw new Error(configuration.reason);
  const token = await getAccessToken();
  if (!token) throw new Error('Microsoft access token unavailable. Sign in again.');
  const url = `${STAFF_BASE}/${path.replace(/^\//, '')}`;
  const response = await fetch(url, {
    method: options.method ?? 'GET',
    headers: { Authorization: `Bearer ${token}`, Accept: 'application/json',
      ...(options.body === undefined ? {} : {'Content-Type': 'application/json'}) },
    body: options.body === undefined ? undefined : JSON.stringify(options.body),
    signal: options.signal,
  });
  if (response.status === 401 || response.status === 403)
    throw new Error('n8n denied access. Confirm Microsoft token audience and staff/course permissions.');
  if (!response.ok) throw new Error(`n8n request failed (HTTP ${response.status}).`);
  const data: unknown = await response.json();
  if (!data || typeof data !== 'object') throw new Error('n8n returned an invalid JSON response.');
  return data as T;
}

export async function loadDashboard(getToken:()=>Promise<string>,signal?:AbortSignal):Promise<DashboardData> {
  if(isDemoMode())return demoDashboard();
  const data=await request<DashboardData>('dashboard',getToken,{signal});
  if (!Array.isArray(data.courses)||!Array.isArray(data.trend)||data.source!=='qwickly')throw new Error('Staff dashboard response is not verified Qwickly data.');
  return data;
}
export async function findStudents(q:string,getToken:()=>Promise<string>,signal?:AbortSignal):Promise<StudentSummary[]> {
  if(isDemoMode())return demoStudents().filter(s=>(s.name+' '+s.sisId+' '+s.programme).toLowerCase().includes(q.toLowerCase()));
  if(q.trim().length<3) return []; // Avoid unbounded live roster enumeration.
  const data=await request<{students:StudentSummary[]}>(`students?query=${encodeURIComponent(q)}`,getToken,{signal});
  if (!Array.isArray(data.students)) throw new Error('Student search returned an invalid list.');
  return data.students;
}
export async function listCases(getToken:()=>Promise<string>,signal?:AbortSignal):Promise<StaffCase[]> {
  if(isDemoMode())return demoCases.map(c=>({...c}));
  const data=await request<{cases:StaffCase[]}>('cases',getToken,{signal});
  if (!Array.isArray(data.cases))throw new Error('Case API returned an invalid list.');
  return data.cases;
}
export async function addCase(data:Pick<StaffCase,'sisId'|'studentName'|'reason'|'priority'|'note'>,getToken:()=>Promise<string>):Promise<StaffCase> {
  if(isDemoMode()){
    const entry:StaffCase={...data,id:`DEMO-CASE-${Date.now()}`,status:'open',owner:'Unassigned',createdAt:new Date().toISOString()};
    demoCases=[entry,...demoCases];return {...entry};
  }
  const response=await request<{case:StaffCase}>('cases',getToken,{method:'POST',body:{action:'create',...data}});
  if(!response.case?.id)throw new Error('Case creation was not confirmed by n8n.');
  return response.case;
}
export async function changeCaseStatus(id:string,status:CaseStatus,getToken:()=>Promise<string>):Promise<StaffCase> {
  if(isDemoMode()){
    const current=demoCases.find(c=>c.id===id);if(!current)throw new Error('Case not found.');
    const updated={...current,status};demoCases=demoCases.map(c=>c.id===id?updated:c);return updated;
  }
  const response=await request<{case:StaffCase}>('cases',getToken,{method:'POST',body:{action:'updateStatus',id,status}});
  if(response.case?.status!==status)throw new Error('n8n did not confirm the case status update.');
  return response.case;
}
export async function assignCaseToMe(id: string, staffName: string, getToken: () => Promise<string>): Promise<StaffCase> {
  if (isDemoMode()) {
    const current=demoCases.find(c=>c.id===id);
    if(!current)throw new Error('Case not found.');
    const updated={...current,owner:staffName||'Demo Reviewer'};
    demoCases=demoCases.map(c=>c.id===id?updated:c);
    return updated;
  }
  // n8n must derive the assignee from a verified Microsoft token, not from user input.
  const response=await request<{case:StaffCase}>('cases',getToken,{method:'POST',body:{action:'assignToMe',id}});
  if(!response.case?.id)throw new Error('n8n did not confirm the assignment.');
  return response.case;
}

}
