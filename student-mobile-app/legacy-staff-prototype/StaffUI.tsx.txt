export function AppScaffold({title,eyebrow,subtitle,children,action}:{title:string;eyebrow?:string;subtitle?:string;children:ReactNode;action?:ReactNode}) {
  const router=useRouter();const path=usePathname();const {user}=useAuth();const config=staffApiConfig();
  return <SafeAreaView edges={['top','bottom']} style={styles.safe}>
    <View style={styles.top}>
      <BrandLogo width={77}/>
      <View style={styles.topDivider}/>
      <View style={{flex:1}}><Text style={styles.brandName}>Attendance Intelligence</Text><Text style={styles.brandSub}>Swinburne Sarawak · FYP prototype</Text></View>
      <View style={[styles.modeDot,{backgroundColor:config.enabled?brand.positive:brand.amber}]}/>
    </View>
    <ScrollView keyboardShouldPersistTaps="handled" style={styles.scroll} contentContainerStyle={styles.content}>
      <View style={styles.headingRow}>
        <View style={{flex:1}}>{eyebrow?<Text style={styles.eyebrow}>{eyebrow.toUpperCase()}</Text>:null}<Text style={styles.pageTitle}>{title}</Text>{subtitle?<Text style={styles.subtitle}>{subtitle}</Text>:null}</View>{action}
      </View>
      {!config.enabled && <View style={styles.demoStrip}><View style={styles.demoCircle}/><Text style={styles.demoText}>DEMO PREVIEW · Fictional data, not official Qwickly records</Text></View>}
      {user?.method==='development' && config.enabled && <View style={styles.warning}><Text style={styles.demoText}>Development sign-in has no Microsoft token; live staff requests are disabled.</Text></View>}
      {children}
      <Text style={styles.foot}>FYP research prototype · Not an official Swinburne service or compliance determination.</Text>
    </ScrollView>
    <View style={styles.nav}>{tabs.map(t=>{const active=path===t.path || (t.path==='/students'&&path.startsWith('/student-detail')) || (t.path==='/more'&&path==='/audit');return <Pressable key={t.path} accessibilityRole="tab" accessibilityState={{selected:active}} onPress={()=>router.navigate(t.path)} style={styles.navItem}><View style={active?styles.activeIcon:undefined}><Icon name={t.icon} size={21} color={active?brand.red:brand.muted}/></View><Text style={[styles.navLabel,active&&styles.navSelected]}>{t.label}</Text></Pressable>;})}</View>
  </SafeAreaView>;
}
